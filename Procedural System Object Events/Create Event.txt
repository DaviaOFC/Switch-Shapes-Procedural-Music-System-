function transform_riff(_riff, _mode) {
    var _n   = array_length(_riff);
    var _out = array_create(_n, -1);
    var _max = array_length(notes_scale) - 1;

    switch (_mode) {
        case "transpose": {
            var _k = choose(-2, -1, 1, 2);
            for (var i = 0; i < _n; i++)
                _out[i] = (_riff[i] >= 0) ? clamp(_riff[i] + _k, 0, _max) : -1;
            break;
        }
        case "invert": {
            var _axis = 5;
            for (var i = 0; i < _n; i++) {
                if (_riff[i] >= 0) { _axis = _riff[i]; break; }
            }
            for (var i = 0; i < _n; i++)
                _out[i] = (_riff[i] >= 0) ? clamp(_axis - (_riff[i] - _axis), 0, _max) : -1;
            break;
        }
        case "retrograde": {
            for (var i = 0; i < _n; i++) _out[i] = _riff[_n - 1 - i];
            break;
        }
        case "rotate": {
            var _shift = choose(1, 2);
            for (var i = 0; i < _n; i++) _out[i] = _riff[(i + _shift) % _n];
            break;
        }
        case "thin": {
            for (var i = 0; i < _n; i++) _out[i] = (i % 2 == 0) ? _riff[i] : -1;
            break;
        }
        case "fill": {
            for (var i = 0; i < _n; i++) {
                if (_riff[i] >= 0) {
                    _out[i] = _riff[i];
                } else {
                    var _neighbor = -1;
                    for (var j = i - 1; j >= 0; j--) {
                        if (_riff[j] >= 0) { _neighbor = _riff[j]; break; }
                    }
                    _out[i] = (_neighbor >= 0) ? clamp(_neighbor + choose(-1, 1), 0, _max) : -1;
                }
            }
            break;
        }
        default: {
            array_copy(_out, 0, _riff, 0, _n);
        }
    }
    return _out;
}

        if (random(1) > 0.5) riff_rhythm[3] = 1;
    }
    
    
    if (tension_factor > 0.6 && random(1) > 0.6) {
        riff_rhythm[1] = 1; 
    }

    
    for (var bar = 0; bar < 4; bar++) {
        var bar_start = bar * 4;
        var is_answer_bar = (bar == 3);

        for (var k = 0; k < 4; k++) {
            var step = bar_start + k;
            var note = -1;
            var play = false; 

            if (is_answer_bar) {
                var question_note = current_melody_riff[3 - k]; 
                play = (question_note >= 0) && (riff_rhythm[k] == 1 || k == 3); 
                note = question_note;

                if (play) { note = clamp(note - 2, 0, max_idx); }
                if (k == 3) { note = 0; play = true; } 
                
            } else if (bar == 1 && k == 3) {
                note = current_melody_riff[k];
                play = (note >= 0) && (riff_rhythm[k] == 1);
                
                if (random(1) < melody_fill_prob && !play) {
                    play = true;
                    note = current_melody_riff[0]; 
                }
            } else {
                note = current_melody_riff[k];
                play = (note >= 0) && (riff_rhythm[k] == 1);
            }
            
            
            
            
            
            if (!play && tension_factor > 0.7 && random(1) > 0.8) {
                play = true;
                note = current_melody_riff[clamp(k - 1, 0, 3)]; 
            }
            
            
            if (play && k == 3 && tension_factor > 0.4) {
                if (random(1) > 0.7) { 
                    note = clamp(note + choose(-1, 1), 0, max_idx); 
                }
            }

            
            if (play && note >= 0) {
                _mel_act[step] = 1;
                _mel_idx[step] = clamp(note, 0, max_idx);
                
                
                var human_vel = (k == 0) ? random_range(0.9, 1.0) : random_range(0.55, 0.85);
                
                if (tension_factor > 0.6 && random(1) > 0.8) {
                    human_vel = choose(0.2, 1.2); 
                }
                
                _mel_vel[step] = human_vel;
            }
        }
    }
		
		if (is_section_b && random(1) < 0.35) {
    var _rest_bar = choose(1, 2);
    for (var i = _rest_bar * 4; i < (_rest_bar + 1) * 4; i++) {
						track_melody_active[i] = 0;
				}
		}
		
		for (var bar = 0; bar < 4; bar++) {
    var _has_melody = false;
    for (var k = 0; k < 4; k++) {
        if (track_melody_active[bar*4 + k]) { _has_melody = true; break; }
    }
    if (!_has_melody) {
        track_snare[bar*4 + 2] = max(track_snare[bar*4 + 2], 0.7);
        track_gongue[bar*4 + 3] = max(track_gongue[bar*4 + 3], 0.8);
    }
}
    
    var bass_choice;
    switch (pattern_mood) {
        case "tribal":
            bass_choice = choose(bass_patterns_tribal[0], bass_patterns_tribal[1], bass_patterns_tribal[2]); break;
        case "dub":
            bass_choice = choose(bass_patterns_dub[0], bass_patterns_dub[1], bass_patterns_dub[2]); break;
        case "aggressive":
            bass_choice = choose(bass_patterns_aggressive[0], bass_patterns_aggressive[1], bass_patterns_aggressive[2]); break;
        case "minimal":
            bass_choice = choose(bass_patterns_minimal[0], bass_patterns_minimal[1]); break;
    }

    for (var i = 0; i < steps_per_loop; i++) {
        _bass[i] = bass_choice[i];
        
        
        
        if (_bass[i] == 0 && (i % 2 != 0) && pattern_mood != "minimal") {
            
            if (random(1) < (0.15 * tension_factor)) {
                _bass[i] = 2; 
            }
        }

        
        if (_bass[i] > 0) {
            if (_kick[i] == 1.0) {
                if (random(1) < 0.30) _bass[i] = 2; 
            } else if (_kick[i] > 0 && _kick[i] < 1.0) {
                _bass[i] = 0.5; 
            }
        }
        
        
        
        if (i >= 13) {
            if (random(1) < (0.35 * tension_factor)) {
                _bass[i] = choose(1, 0.5, 2); 
            }
        }
    }

    
    var arp_bank;
    switch (pattern_mood) {
        case "tribal":     arp_bank = arp_sequences_tribal;     break;
        case "dub":        arp_bank = arp_sequences_dub;        break;
        case "aggressive": arp_bank = arp_sequences_aggressive; break;
        case "minimal":    arp_bank = arp_sequences_minimal;    break;
    }
    var _orig = arp_bank[irandom(array_length(arp_bank)-1)];
	
	var arp_seq = array_create(array_length(_orig));
	
	array_copy(arp_seq, 0, _orig, 0, array_length(_orig));
    if (current_melody_riff[0] >= 0) {
    var safe_idx = clamp(current_melody_riff[0], 0, array_length(notes_scale)-1);
    arp_seq[0] = notes_scale[safe_idx]; 
}
    if (random(1) < 0.25) {
		var mut_idx = irandom(3);
		var safe_notes = [0, 3, 5, 7, 10, 12]; 
		arp_seq[mut_idx] = safe_notes[irandom(array_length(safe_notes) - 1)]; 
	}

    var arp_density = 0.5 + (tension_factor * 0.5);
    
    var arp_overclock = (pattern_mood == "aggressive" || tension_factor > 0.7);

    for (var i = 0; i < steps_per_loop; i++) {
        if (arp_active) {
            var is_pulse = (i % 4 == 0);
            
            if (arp_overclock || is_pulse || random(1) < arp_density) {
                _arp[i]   = is_pulse ? 1.0 : random_range(0.6, 0.85);
                _arp_n[i] = arp_seq[i % 4];
            } else {
                _arp[i] = 0; _arp_n[i] = 0;
            }
        } else {
            _arp[i] = 0; _arp_n[i] = 0;
        }
    }

    
var _breque = ((pattern_mood == "aggressive" || pattern_mood == "tribal") && random(1) < 0.20);
if (_breque) {
    for (var i = 8; i <= 13; i++) {
        _kick[i]=0; _snare[i]=0; _hat[i]=0; _bass[i]=0;
        _mel_act[i]=0; _arp[i]=0; _gongue[i]=0;
    }
    _gongue[9] = 1.0; _gongue[11] = 1.0;
    _snare[14] = 0.3;
    _kick[15] = 1.0; _snare[15] = 1.0; _bass[15] = 2; _gongue[15] = 1.0;
}
if (!is_preview) { is_breque_active = _breque; is_breque_exposed = _breque; }
	
	if (pattern_mood == "dub" || pattern_mood == "tribal") {
        if (random(1) < 0.15) { 
            dub_siren_pitch = random_range(4.0, 6.0); 
            dub_siren_vol = 1.0;
        }
    }
	
	if (is_preview) {
    next_state = {
        mood: pattern_mood, scale: notes_scale,
        arp_active: arp_active, echo_chance: snare_echo_chance,
        mk: master_kick, ms: master_snare, mh: master_hat, mb: master_bass,
        mm: master_melody, ma: master_arp, mn: master_noise,
        breque: _breque
    };
}
	
	if (is_preview && _sv != undefined) {
		pattern_mood            = _sv.mood;
		notes_scale             = _sv.scale;
		last_melody_mood        = _sv.last_melody_mood;
		melody_riff_age         = _sv.melody_riff_age;
		current_melody_riff     = _sv.riff;
		mood_bias               = _sv.bias;
		mood_bias_strength      = _sv.bias_str;
		master_kick             = _sv.mk;
		master_snare            = _sv.ms;
		master_hat              = _sv.mh;
		master_bass             = _sv.mb;
		master_melody           = _sv.mm;
		master_arp              = _sv.ma;
		master_noise            = _sv.mn;
		mood_arp_active         = _sv.arp_active;
		mood_snare_echo_chance  = _sv.echo_chance;
		is_breque_active        = _sv.is_breque;
		is_breque_exposed       = _sv.is_breque_exp;
	}
}
