					master_melody = next_state.mm;
					master_arp = next_state.ma;
					master_noise = next_state.mn;
					is_breque_active = next_state.breque;
					is_breque_exposed = next_state.breque;
					next_state = undefined;
				}
        next_pattern_ready = false;
    } else {
        generate_pattern();
    }
}

    var moods = ["tribal", "dub", "aggressive", "minimal"];
    for (var i = 0; i < 4; i++) {
        var m = moods[i];
        var current = ds_map_find_value(mood_pressure, m);
        var decayed = current * mood_pressure_decay;
        if (decayed < 0.5) decayed = 0;  
        ds_map_set(mood_pressure, m, decayed);
    }
}

if (current_step == 12 && !next_pattern_ready) {
    generate_pattern(true);
    next_pattern_ready = true;

    for (var i = 13; i < steps_per_loop; i++) {
        if (track_arp_next[i] == 1 && track_arp[i] == 0) {
            if (random(1) < 0.35) {
                track_arp[i]      = 1;
                track_arp_note[i] = track_arp_note_next[i];
            }
        }
    }
}

}

if (global.volume_music > 0) {

    sidechain_envelope = lerp(sidechain_envelope, 1.0, 0.15); 

        if (noise_snd != -1) {
        if (!audio_is_playing(noise_snd)) {
            var ns = audio_play_sound(noise_snd, 1, true);
            noise_snd = ns;
        }
        
        var base_noise_vol = 0.0;
        if (pattern_mood == "aggressive") base_noise_vol = 0.85;
        else if (pattern_mood == "dub") base_noise_vol = 0.65;
        else if (pattern_mood == "tribal") base_noise_vol = 0.50;
        else base_noise_vol = 0.35;
        
        if (tension_factor > 0.75) base_noise_vol += 0.3;
        
        var master = global.volume_music / 4;
        noise_vol = lerp(noise_vol, base_noise_vol * master, 0.03);
        
		if (!audio_is_playing(noise_snd)) {
			var ns = audio_play_sound_on(foley_emitter, snd_noise_vinyl, true, 1);
			noise_snd = ns;
		}

		var timer = current_time / 1000.0;
		var velocidade_orbita = 0.5;

		var auto_pan_x = sin(timer * velocidade_orbita) * 150;
		var auto_pan_z = cos(timer * velocidade_orbita) * 150;
		audio_emitter_position(foley_emitter, auto_pan_x, 0, auto_pan_z);

		audio_sound_gain(noise_snd, noise_vol * master_noise * sidechain_envelope, 0);
		var radio_pitch = 1.0; 
        
        if (tape_speed < 0.9 || (pattern_mood == "dub" && random(1) > 0.98)) {
            
            radio_pitch = choose(0.2, 0.5, 2.5, 3.0) * random_range(0.9, 1.1);
        }
        
        audio_sound_pitch(noise_snd, radio_pitch * tape_speed);
    
    if (!audio_is_playing(space_drone_snd)) {
        var drone = audio_play_sound_on(space_emitter, snd_newsynth_od2, true, 2);
        space_drone_snd = drone;
    }

    var orbit_timer = current_time / 4500.0; 
    var doppler_timer = current_time / 2000.0; 
    var breath_timer = current_time / 3000.0; 
    
    var space_x = sin(orbit_timer) * 150; 
    var space_z = cos(orbit_timer) * 150; 
    audio_emitter_position(space_emitter, space_x, 0, space_z);
    
    var space_pitch = 2.3 + (cos(doppler_timer) * 0.15); 
    
    audio_sound_pitch(space_drone_snd, space_pitch * tape_speed); 
    
    var vol_osc = (sin(breath_timer) + 1.0) / 2.0; 
    var base_space_vol = 0.35 * (global.volume_music / 4); 
    
    if (pattern_mood == "aggressive") {
        base_space_vol *= 0.1;
    } else if (pattern_mood == "dub") {
        base_space_vol *= 1.4; 
    }
    
    audio_sound_gain(space_drone_snd, vol_osc * base_space_vol * sidechain_envelope, 0);
    }

    if (dub_siren_vol > 0.01) {
        
        dub_siren_pitch = lerp(dub_siren_pitch, 0.5, 0.04); 
        dub_siren_vol = lerp(dub_siren_vol, 0, 0.03); 

        var siren_pan = sin(current_time / 100) * 0.8;
        var snd_sir = play_panned(snd_newsynth, 9, false, siren_pan);
               
        audio_sound_gain(snd_sir, dub_siren_vol * 0.25 * (global.volume_music / 4) * sidechain_envelope, 0);
        audio_sound_pitch(snd_sir, dub_siren_pitch);
    }
